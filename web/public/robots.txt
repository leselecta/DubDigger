# Dub Digger — https://dubdigger.com
#
# The corpus is 420,575 artists and 113,952 labels, and every one of those
# pages is a live query against a 928 MB SQLite file on one small VPS. So this
# file is a load rule as much as an indexing one.

User-agent: *
Allow: /

# Every list carries its state in the query string: ?tab=, ?sort=, ?show=, and
# ?q= on the home page. Those are the same page answering the same question in
# a different order, which is what the canonical already says. This keeps a
# crawler from walking every combination to find that out.
Disallow: /*?

# The dropdown's partial. It answers on a clean URL as well as with a query,
# and what it returns is a fragment: rows of markup with no <head> on them, so
# no title, no canonical and nothing saying what it is. The rule above already
# covers /suggest?q=, which is the only way anything calls it. This covers the
# bare path, which nothing links to and a crawler should still never file as a
# page of this site.
Disallow: /suggest

# Ignored by Google, honoured by Bing, Yandex and Seznam. Five seconds is
# plenty for a site whose sitemap is two thousand URLs.
Crawl-delay: 5

Sitemap: https://dubdigger.com/sitemap.xml
